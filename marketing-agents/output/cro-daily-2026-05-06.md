### Re-escalating last week's `<NewsletterPopup />` spec (carry from 4/29)

`web/src/components/NewsletterPopup.tsx` (new):

```tsx
'use client';

import { useState, useEffect, useRef } from 'react';
import NewsletterSignup from '@/components/NewsletterSignup';

type Trigger = 'exit-intent' | 'time' | 'scroll-depth';

interface Props {
  trigger: Trigger;
  delaySeconds?: number;     // for trigger='time'
  scrollPercent?: number;    // for trigger='scroll-depth'
  source: string;            // for newsletter_subscribers.source attribution
  storageKey?: string;       // dismissal persistence key (default: 'newsletter_popup_dismissed')
  cooldownDays?: number;     // default 7
  headline?: string;
  subhead?: string;
}

export default function NewsletterPopup({
  trigger,
  delaySeconds = 30,
  scrollPercent = 50,
  source,
  storageKey = 'newsletter_popup_dismissed',
  cooldownDays = 7,
  headline = 'Wait — get one ticket-saving tip before you go',
  subhead = 'Weekly: exclusive deals, price-drop alerts, and savings on the events you care about.',
}: Props) {
  const [show, setShow] = useState(false);
  const fired = useRef(false);

  useEffect(() => {
    if (typeof window === 'undefined') return;

    // Cooldown check
    const dismissedAt = localStorage.getItem(storageKey);
    if (dismissedAt) {
      const ageMs = Date.now() - parseInt(dismissedAt, 10);
      if (ageMs < cooldownDays * 24 * 60 * 60 * 1000) return;
    }

    if (trigger === 'exit-intent') {
      const handler = (e: MouseEvent) => {
        if (e.clientY < 10 && !fired.current) {
          fired.current = true;
          setShow(true);
          pushDataLayer({ event: 'newsletter_popup_shown', trigger, source });
        }
      };
      document.addEventListener('mouseleave', handler);
      return () => document.removeEventListener('mouseleave', handler);
    }

    if (trigger === 'time') {
      const t = setTimeout(() => {
        if (!fired.current) {
          fired.current = true;
          setShow(true);
          pushDataLayer({ event: 'newsletter_popup_shown', trigger, source });
        }
      }, delaySeconds * 1000);
      return () => clearTimeout(t);
    }

    if (trigger === 'scroll-depth') {
      const handler = () => {
        const pct = (window.scrollY + window.innerHeight) / document.documentElement.scrollHeight * 100;
        if (pct >= scrollPercent && !fired.current) {
          fired.current = true;
          setShow(true);
          pushDataLayer({ event: 'newsletter_popup_shown', trigger, source });
        }
      };
      window.addEventListener('scroll', handler, { passive: true });
      return () => window.removeEventListener('scroll', handler);
    }
  }, [trigger, delaySeconds, scrollPercent, source, storageKey, cooldownDays]);

  const handleDismiss = () => {
    setShow(false);
    if (typeof window !== 'undefined') {
      localStorage.setItem(storageKey, String(Date.now()));
    }
    pushDataLayer({ event: 'newsletter_popup_dismissed', trigger, source });
  };

  if (!show) return null;

  return (
    <div
      role="dialog"
      aria-modal="true"
      aria-labelledby="newsletter-popup-title"
      onKeyDown={(e) => { if (e.key === 'Escape') handleDismiss(); }}
      className="fixed inset-0 bg-black/60 flex items-center justify-center z-50 p-4"
      onClick={handleDismiss}
    >
      <div
        onClick={(e) => e.stopPropagation()}
        className="bg-white rounded-2xl max-w-md w-full p-6 shadow-2xl relative"
      >
        <button
          onClick={handleDismiss}
          aria-label="Close"
          className="absolute top-3 right-3 text-gray-400 hover:text-gray-600 text-xl leading-none"
        >
          ×
        </button>
        <h3 id="newsletter-popup-title" className="text-xl font-heading font-bold text-navy mb-2">
          {headline}
        </h3>
        <p className="text-gray-500 text-sm mb-4">{subhead}</p>
        <NewsletterSignup variant="popup" source={source} />
      </div>
    </div>
  );
}

function pushDataLayer(payload: Record<string, unknown>) {
  if (typeof window === 'undefined') return;
  const w = window as { dataLayer?: Record<string, unknown>[] };
  w.dataLayer = w.dataLayer ?? [];
  w.dataLayer.push(payload);
}
```
